We build "|||" entries, sort, reverse, then iterate. {%- endcomment -%} {% assign cate_entries = "" | split: "" %} {% for category in site.categories %} {% assign latest_post = category[1] | first %} {% capture cate_entry %}{{ latest_post.date | date: '%s' }}|||{{ category[0] }}{% endcapture %} {% assign cate_one = cate_entry | split: "$$NOPE$$" %} {% assign cate_entries = cate_entries | concat: cate_one %} {% endfor %} {% assign cate_entries = cate_entries | sort | reverse %}
{% for entry in cate_entries %} {% assign parts = entry | split: "|||" %} {% assign cate_name = parts[1] %} {% assign cate_posts = site.categories[cate_name] %} {% endfor %}
